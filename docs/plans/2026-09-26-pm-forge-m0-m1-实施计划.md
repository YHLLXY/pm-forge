```python
        'category_id': 'BIGINT', 'behavior_type': 'VARCHAR', 'ts': 'BIGINT'}
VALID_BEHAVIORS = {'pv', 'cart', 'fav', 'buy'}


def validate_userbehavior(path: str) -> dict:
    rel = duckdb.read_csv(path, header=False, columns=COLS)
    rows = rel.count("*").fetchone()[0]
    bad = rel.project("count(*) FILTER (behavior_type NOT IN ('pv','cart','fav','buy'))").fetchone()[0]
    return {"rows": rows, "bad_behavior_rows": bad, "columns_ok": True}


def sample_events(src: str, dst_parquet: str, n: int = 100_000, seed: int = 42) -> int:
    rel = duckdb.read_csv(src, header=False, columns=COLS)
    # 确定性抽样：先小样验证 REPEATABLE 语法（见 Step 3 验证步），失败则走 pandas 兜底
    try:
        out = rel.project(f"* USING SAMPLE reservoir({n} ROWS) REPEATABLE ({seed})")
        df = out.df()
    except (duckdb.Error, Exception):
        df = rel.df().sample(n=n, random_state=seed)
    df.to_parquet(dst_parquet, index=False)
    return len(df)


def to_datetime_cn(s: pd.Series) -> pd.Series:
    return pd.to_datetime(s, unit="s", utc=True).dt.tz_convert("Asia/Shanghai")
```

```python
def compute_funnel(df, stages):
    rows = []
    prev_users = None
    for i, behaviors in enumerate(stages):
        users = df[df["behavior_type"].isin(behaviors)]["user_id"].nunique()
        rate = None if prev_users in (None, 0) else users / prev_users
        rows.append({"stage": i, "behaviors": "+".join(behaviors), "users": users, "conv_rate": rate})
        prev_users = users
    return pd.DataFrame(rows)
```
